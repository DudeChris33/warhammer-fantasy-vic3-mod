### For social monarchies. This exists so that the various monarch titles are preserved after instating a social monarchy.
gov_social_monarchy_empire = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_EMPEROR"
	female_ruler = "RULER_TITLE_EMPRESS"

	male_heir = "RULER_TITLE_PRINCE"
	female_heir = "RULER_TITLE_PRINCESS"

	possible = {
		has_law = law_type:law_social_monarchy
		coa_fascist_trigger = no
		coa_absolute_monarchy_trigger = no
		OR = {
			country_tier = hegemony
			country_tier = empire
		}
		has_gov_regency = no
	}

	on_government_type_change = {
		change_to_hereditary = yes
	}
	on_post_government_type_change = {
		post_change_to_hereditary = yes
	}
}
