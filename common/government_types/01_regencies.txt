### For regencies. Non-regency monarchies are split off into government_types/01_monarchies
gov_regency_empire = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_REGENT"
	female_ruler = "RULER_TITLE_REGENT"

	male_heir = "RULER_TITLE_EMPEROR"
	female_heir = "RULER_TITLE_EMPRESS"

	possible = {
		country_has_monarchy_law = yes
		OR = {
			country_tier = hegemony
			country_tier = empire
		}
		has_gov_regency = yes
		NOT = { has_law = law_type:law_technocracy }
		country_has_voting_franchise = no
	}

	on_government_type_change = {
		change_to_hereditary = yes
	}
	on_post_government_type_change = {
		post_change_to_hereditary = yes
	}
}
