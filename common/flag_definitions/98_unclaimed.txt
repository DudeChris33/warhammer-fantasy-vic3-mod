# common/flag_definitions is a replace_paths folder. This mod's DEFAULT list
# (00_flag_definitions.txt) ships one unconditionally-applicable block at
# priority = 0 — so "no flag definition is applicable for a country" (vanilla's
# own tag-as-COA_KEY fallback condition, see that file's vanilla header comment
# at .../game/common/flag_definitions/00_flag_definitions.txt:1-14) never fires
# for any tag. Without a higher-priority tag-named list here, the placeholder COA
# in common/coat_of_arms/coat_of_arms/98_unclaimed.txt (board #6391) would parse
# but sit unused behind default_flag. The list below is a single unconditional
# block at priority = 100 — above the priority = 0 DEFAULT, below vanilla's own
# conditional range of 100-2000 for future per-tag variants.
# `always = yes` confirmed valid at docs/triggers.log.

UNO = {
	flag_definition = {
		coa = UNO
		priority = 100
		trigger = { always = yes }
	}
}
