# The two effect calls and the constitutionalist -> republican ideology swap are verbatim;
# both ideologies ship in common/ideologies/.
# common/journal_entries is blanked via replace_paths and je_abolish_monarchy is undefined.
# has_journal_entry on a missing journal-entry type is a HARD LOAD ERROR, so vanilla's
# `NOR = { has_journal_entry = je_abolish_monarchy has_variable = monarchy_stamped_out_var }`
# guard is narrowed to the has_variable half alone. add_journal_entry on a missing type is
# only silently inert rather than an error, but the `add_journal_entry = { type =
# je_abolish_monarchy }` line is dropped too, for hygiene.
# DISCLOSED GAP: no journal entry tracks the abolition of monarchy, and the guard is now
# weaker than vanilla's - a country that has already begun abolishing the monarchy is no
# longer detected by its journal entry, only by monarchy_stamped_out_var, so
# abolishing_monarchy_var can be re-zeroed in a case vanilla would have skipped.
# [#6573] monarchy_stamped_out_var is only set by real-world content that was never
# restored, so its read below is neutralized and the guard is always true here; the
# abolishing_monarchy_var writes (here and in common/on_actions/) were removed because
# nothing reads it until the abolition JE returns.
# A follow-up should author a WH-native je_abolish_monarchy and restore the full guard.
transition_off_of_monarchy_effects = {
	crown_land_republic_effect = yes
	transition_off_of_monarchy_law_effects = yes
	if = {
		limit = {
			any_interest_group = {
				has_ideology = ideology:ideology_constitutionalist
			}
		}
		every_interest_group = {
			limit = {
				has_ideology = ideology:ideology_constitutionalist
			}
			remove_ideology = ideology_constitutionalist
			add_ideology = ideology_republican
		}
	}
	if = {
		limit = {
			NOT = { always = no }  # [#6573] was: NOT = { has_variable = monarchy_stamped_out_var }
		}
		# [#6573] removed: set_variable = {
			# [#6573] removed: name = abolishing_monarchy_var
			# [#6573] removed: value = 0
		# [#6573] removed: }
	}
}
