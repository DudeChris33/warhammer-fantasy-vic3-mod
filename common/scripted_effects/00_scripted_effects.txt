save_agitator_law_scopes = {
	if = {
		limit = {
			agitator_supports_enaction = yes
		}
		random_scope_character = {
			limit = {
				has_role_of_type = agitator
				OR = {
					law_enactment_stance = {
						law = owner.currently_enacting_law.type
						value > neutral
					}
					AND = {
						law_enactment_stance = {
							law = owner.currently_enacting_law.type
							value = neutral
						}
						interest_group ?= {
							law_enactment_stance = {
								law = owner.currently_enacting_law.type
								value > neutral
							}
						}
					}
				}
			}
			save_scope_as = supporting_agitator
			interest_group ?= {
				save_scope_as = agitator_ig
			}
		}
		random_political_movement = {
			limit = {
				law_enactment_stance = {
					law = owner.currently_enacting_law.type
					value > neutral
				}
			}
			save_scope_as = supporting_movement
		}
	}

	if = {
		limit = {
			agitator_opposes_enaction = yes
		}
		random_scope_character = {
			limit = {
				has_role_of_type = agitator
				OR = {
					law_enactment_stance = {
						law = owner.currently_enacting_law.type
						value < neutral
					}
					AND = {
						law_enactment_stance = {
							law = owner.currently_enacting_law.type
							value = neutral
						}
						interest_group ?= {
							law_enactment_stance = {
								law = owner.currently_enacting_law.type
								value < neutral
							}
						}
					}
				}
			}
			save_scope_as = opposing_agitator
			interest_group ?= {
				save_scope_as = agitator_ig
			}
		}
		random_political_movement = {
			limit = {
				law_enactment_stance = {
					law = owner.currently_enacting_law.type
					value < neutral
				}
			}
			save_scope_as = opposing_movement
		}

# Used by the generic prestige-good JEs in 05_prestige_goods.txt. If the
# company currently saved to scope:relevant_company no longer exists
# (e.g. the player disbanded it), rebind scope:relevant_company to another
# qualifying company in the country and reset the progress timer. If no
# qualifying company exists, leaves the scope unbound — the JE's possible
# trigger will then fail and can_deactivate = yes sends it back to inactive.
rebind_prestige_good_company = {
	if = {
		limit = { NOT = { exists = scope:relevant_company } }
		random_company = {
			limit = {
				can_potentially_produce_prestige_goods = $GOODS$
				is_producing_prestige_goods = no
				company_is_prosperous = yes
				NOT = { has_variable = $UNLOCK_VAR$ }
			}
			save_scope_as = relevant_company
		}
		set_variable = {
			name = $TIMER_VAR$
			value = 0
		}
	}
}

# Used from on_company_disbanded for a single generic prestige good. When the
# company being disbanded (scope:company) is the one that had unlocked $JE$'s
# good, its unlock variable is lost with it. If no other company in the country
# still holds that unlock, re-add the matching prestige-good JE so a future
# qualifying company can earn the good again. The disbanded company still exists
# and still holds the variable when this runs, so count >= 2 means another holder
# remains and the good is not actually lost. Runs in country scope.
re_add_prestige_good_je_if_lost = {
	if = {
		limit = {
			scope:company ?= {
				has_variable = $UNLOCK_VAR$
			}
			NOT = {
				any_company = {
					count >= 2
					has_variable = $UNLOCK_VAR$
				}
			}
			NOT = {
				has_journal_entry = $JE$
			}
		}
		add_journal_entry = {
			type = $JE$
		}
	}
}
