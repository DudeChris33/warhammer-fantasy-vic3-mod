# CLEAN PORT (#3537) - vanilla body verbatim.
# vanilla: 00_victoria_ip4_scripted_effects.txt:1878
# Depends only on content this mod ships: active_law:lawgroup_distribution_of_power,
# amendment_type:amendment_electoral_clientelism (common/amendments/00_amendments_historical_04.txt)
# and modifier:country_forbid_electoral_fraud_bool
# (common/modifier_type_definitions/12_ip4_script_modifiers.txt).
handle_possible_absent_caciquismo_variables = {
	if = {
		limit = {
			active_law:lawgroup_distribution_of_power ?= {
				has_amendment = amendment_type:amendment_electoral_clientelism
			}
			NOT = {
				has_variable = caciquismo_var
			}
		}
		set_variable = caciquismo_var
	}
	if = {
		limit = {
			modifier:country_forbid_electoral_fraud_bool = no
			NOT = {
				active_law:lawgroup_distribution_of_power ?= {
					has_amendment = amendment_type:amendment_electoral_clientelism
				}
			}
			NOT = {
				has_variable = electoral_fraud_abolish_var
			}
		}
		set_variable = { name = electoral_fraud_abolish_var value = 0 }
	}
}

# CLEAN PORT (#3537) - vanilla body verbatim.
# vanilla: 00_victoria_ip4_scripted_effects.txt:1196
# Both ideologies are shipped by this mod (common/ideologies/00_ig_ideologies.txt), and the
# custom_tooltip key switch_from_colonial_slavery_to_pro_slavery_tt resolves from vanilla's
# own localization/english/laws_l_english.yml, which this mod does not override - so no
# localization work is owed here.
switch_from_colonial_slavery_to_pro_slavery = {
	save_temporary_scope_as = landowners_scope
	custom_tooltip = {
		text = switch_from_colonial_slavery_to_pro_slavery_tt
		if = {
			limit = {
				has_ideology = ideology:ideology_pro_slavery_colonial
			}
			remove_ideology = ideology_pro_slavery_colonial
		}
		if = {
			limit = {
				NOT = { has_ideology = ideology:ideology_pro_slavery }
			}
			add_ideology = ideology_pro_slavery
		}
	}
}

# PARTIAL PORT (#3537). vanilla: 00_victoria_ip4_scripted_effects.txt:2528
# The initial set_electoral_confidence and the closing set_variable are verbatim.
# modifier_recent_election_initialisation is undefined in this mod (common/static_modifiers
# has no such entry, and that folder is in replace_paths so vanilla's copy does not load).
# has_modifier / add_modifier / remove_modifier on an undefined named modifier is a HARD
# LOAD ERROR, so those three calls are dropped.
# In vanilla those three calls are the ENTIRE body of the
# `game_date > 1836.1.1 / country_has_voting_franchise = yes / eagle_burger_freedom_index < 0.8`
# branch, so once they are dropped nothing is left for that branch to do and the branch
# itself is dropped rather than left as an empty `if` (see the note in the item report -
# the plan assumed the branch also carried set_electoral_confidence / set_variable calls;
# vanilla's does not). The gate trigger itself is fine: country_has_voting_franchise is
# defined in common/scripted_triggers/00_scripted_triggers.txt.
# DISCLOSED GAP: the decaying election-initialisation legitimacy penalty applied to
# low-freedom-index countries starting after 1836.1.1 is not applied at all. A follow-up may
# restore modifier_recent_election_initialisation to common/static_modifiers/ and, with it,
# this branch.
initialize_electoral_confidence = {
	set_electoral_confidence = {
		value = 1
		multiply = eagle_burger_freedom_index
	}
	set_variable = electoral_confidence_initialized
}
