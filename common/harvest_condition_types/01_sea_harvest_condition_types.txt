# Positive
calm_waters = {
	area_type = sea
	has_animation = no

	trigger = {
		always = yes
	}

	incompatible_with = rough_waters
	incompatible_with = cyclone
	incompatible_with = tropical_storm
	incompatible_with = strong_winds

	range = {
		integer_range = {
			min = 1
			max = 2
		}
	}

	duration = {
		fixed_range = {
			min = 24
			max = 48
		}
	}

	modifier = {
		battle_naval_condition_calm_waters_chance_mult = 1
	}

	intensity = {
		fixed_range = {
			min = 0.5
			max = 1
		}
	}

	chance = {
		value = 0.05
	}

	color = { 100 180 220 }
	icon = "gfx/interface/icons/sea_condition_icons/calm_waters.dds"
	map_texture = "gfx/map/map_painting/sea_conditions_map_patterns/sea_conditions_good.dds"
}

cyclone = {
	area_type = sea
	has_animation = no

	trigger = {
		is_vulnerable_to_tropical_storm = yes
	}

	incompatible_with = calm_waters
	incompatible_with = fog

	range = {
		integer_range = {
			min = 1
			max = 2
		}
	}

	duration = {
		fixed_range = {
			min = 24
			max = 48
		}
	}

	modifier = {
		battle_naval_condition_cyclone_chance_mult = 1
		ship_movement_speed_mult = -0.5
	}

	intensity = {
		fixed_range = {
			min = 0.25
			max = 1
		}
	}

	chance = {
		value = 0
		if = {
			limit = {
				month >= 6
				month <= 11
			}
			add = 0.1
		}
	}

	color = { 80 30 30 }
	icon = "gfx/interface/icons/sea_condition_icons/cyclone.dds"
	map_texture = "gfx/map/map_painting/sea_conditions_map_patterns/sea_conditions_bad.dds"
}
