# Bastion of the Landowners
# Can also appeal to the Devout, PB, and Industrialists

conservative_party = {
	color = { 108 64 179 }
	name = {
		desc = party_conservative
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/conservatism.dds"
	}
	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			OR = {
				py:conservative_party ?= {
					party_has_members = yes
				}
				ig:ig_landowners ?= {
					ig_counts_as_marginal = no
				}
				ig:ig_devout ?= {
					ig_counts_as_marginal = no
				}
				ig:ig_petty_bourgeoisie ?= {
					ig_counts_as_marginal = no
				}
			}
		}
		# unions can never be conservative
		NOT = { is_interest_group_type = ig_trade_unions }
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}
		if = { # socialists and radicals won't join conservative parties
			limit = {
				leader ?= {
					OR = {
						has_socialist_ideology = yes
						has_ideology = ideology:ideology_state_socialist
						has_ideology = ideology:ideology_radical
					}
				}
			}
			add = {
				value = -20
				desc = "from_leftist"
			}
		}
		if = { # conservatives love conservativism
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_traditionalist
						has_ideology = ideology:ideology_royalist
						has_ideology = ideology:ideology_theocrat
						has_ideology = ideology:ideology_slaver
						has_ideology = ideology:ideology_corporatist_leader
					}
				}
			}
			add = {
				value = 5
				desc = "from_conservative_leader"
			}
		}
		if = { # conservatives love conservativism
			limit = {
				OR = {
					has_paternalist_ideology = yes
					has_hierarchic_ideology = yes
					has_patriarchal_ideology = yes
					has_ideology = ideology:ideology_moralist
					has_ideology = ideology:ideology_corporatist
				}
			}
			add = {
				value = 5
				desc = "from_conservative_ig"
			}
		}
		if = { # armed forces will join if a more powerful member has jingoist leader
			limit = {
				is_interest_group_type = ig_armed_forces
				exists = owner.py:conservative_party
				owner = {
					any_interest_group = {
						is_member_of_party = owner.py:conservative_party
						is_powerful = yes
						leader ?= {
							has_ideology = ideology:ideology_jingoist_leader
						}
					}
				}
			}
			add = {
				value = 5
				desc = "from_jingoist_party_leader"
			}
		}
		if = { # landowners will join if they are even moderately powerful
			limit = {
				ig_clout >= 0.15
				is_interest_group_type = ig_landowners
			}
			add = {
				value = 5
				desc = "from_powerful_conservative_ig"
			}
		}
		if = { # various progressives dislike conservativism
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_feminist
						has_ideology = ideology:ideology_abolitionist
						has_ideology = ideology:ideology_pacifist
						has_ideology = ideology:ideology_humanitarian
						has_ideology = ideology:ideology_humanitarian_royalist
					}
				}
			}
			add = {
				value = -5
				desc = "from_progressive_con"
			}
		}
		if = { # reactionaries like conservatism
			limit = {
				has_ideology = ideology:ideology_reactionary
			}
			add = {
				value = 3
				desc = "from_reactionaries"
			}
		}
		# Abolitionists won't join conservative party
		if = {
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_slavery_banned }
				}
				leader ?= {
					has_ideology = ideology:ideology_abolitionist
				}
			}
			add = {
				value = -20
				desc = "from_anti_slavery"
			}
		}
		if = { # patriots are inclined towards conservativism
			limit = {
				OR = {
					has_ideology = ideology:ideology_patriotic
					has_ideology = ideology:ideology_militarist
					has_ideology = ideology:ideology_military_absolutist
				}
			}
			add = {
				value = 5
				desc = "from_patriot"
			}
		}
		if = { # capitalists more likely to join conservatives when the poor can vote
			limit = {
				owner = {
					has_law_or_variant = law_type:law_universal_suffrage
				}
				law_stance = {
					law = law_type:law_universal_suffrage
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_capitalists_suffrage"
			}
		}
		if = { # capitalists and landowners more likely to join with progressive taxation
			limit = {
				owner = {
					has_law_or_variant = law_type:law_graduated_taxation
				}
				law_stance = {
					law = law_type:law_graduated_taxation
					value < neutral
				}
			}
			add = {
				value = 2
				desc = "from_capitalists_tax"
			}
		}
		if = { # capitalists more likely to join conservatives with public health
			limit = {
				owner = {
					has_law_or_variant = law_type:law_public_health_insurance
				}
				law_stance = {
					law = law_type:law_public_health_insurance
					value < neutral
				}
			}
			add = {
				value = 2
				desc = "from_capitalists_health"
			}
		}
		if = { # devout, industrialists more likely to join conservatives with public schools
			limit = {
				owner = {
					has_law_or_variant = law_type:law_public_schools
				}
				law_stance = {
					law = law_type:law_public_schools
					value < neutral
				}
			}
			add = {
				value = 2
				desc = "from_capitalists_schools"
			}
		}
		if = { # devout more likely to join conservatives with total separation
			limit = {
				owner = {
					has_law_or_variant = law_type:law_total_separation
				}
				law_stance = {
					law = law_type:law_total_separation
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_devout_separation"
			}
		}
		if = { # devout more likely to join conservatives with freedom of conscience
			limit = {
				owner = {
					has_law_or_variant = law_type:law_freedom_of_conscience
				}
				law_stance = {
					law = law_type:law_freedom_of_conscience
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_devout_conscience"
			}
		}
		if = { # patriarchs more likely to join conservatives with women's suffrage
			limit = {
				owner = {
					has_law_or_variant = law_type:law_womens_suffrage
				}
				law_stance = {
					law = law_type:law_womens_suffrage
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_patriarchy"
			}
		}
		if = { # patriarchs more likely to join conservatives with working women
			limit = {
				owner = {
					has_law_or_variant = law_type:law_women_in_the_workplace
				}
				law_stance = {
					law = law_type:law_women_in_the_workplace
					value < neutral
				}
			}
			add = {
				value = 2
				desc = "from_patriarchy_lesser"
			}
		}
		if = { # reactionaries like regressing from multiculturalism
			limit = {
				owner = {
					has_law_or_variant = law_type:law_multicultural
				}
				law_stance = {
					law = law_type:law_multicultural
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_multiculturalism"
			}
		}
		# alliance for stricter internal security
		if = {
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_no_home_affairs
						has_law_or_variant = law_type:law_guaranteed_liberties
					}
				}
				law_enactment_stance = {
					law = law_type:law_secret_police
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_liberty"
			}
		}
		if = { # conservatives like censorship
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_right_of_assembly
						has_law_or_variant = law_type:law_protected_speech
					}
				}
				law_enactment_stance = {
					law = law_type:law_censorship
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_no_censorship"
			}
		}
		if = { # conservatives hate republics
			limit = {
				owner = {
					NOR = {
						country_has_monarchy_law = yes
						has_law_or_variant = law_type:law_theocracy
					}
				}
				law_enactment_stance = {
					law = law_type:law_monarchy
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_republic"
			}
		}
		if = { # landowners and industrialists hate labor rights
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_regulatory_bodies
						has_law_or_variant = law_type:law_worker_protections
					}
				}
				law_enactment_stance = {
					law = law_type:law_no_workers_rights
					value > neutral
				}
			}
			add = {
				value = 5
				desc = "from_labor_rights"
			}
		}
		if = { # conservatives hate welfare
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_wage_subsidies
						has_law_or_variant = law_type:law_old_age_pension
					}
				}
				law_enactment_stance = {
					law = law_type:law_poor_laws
					value > neutral
				}
			}
			add = {
				value = 5
				desc = "from_welfare"
			}
		}
		if = { # conservatives like police
			limit = {
				owner = {
					has_technology_researched = law_enforcement
					has_law_or_variant = law_type:law_no_police
				}
				law_stance = {
					law = law_type:law_no_police
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_no_police"
			}
		}
		if = {
			limit = {
				is_interest_group_type = ig_landowners
				is_powerful = yes
			}
			multiply = {
				value = 2
				desc = "from_archetype_party"
			}
		}
		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				ig_counts_as_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = {
				scope:number = 2
			}
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number >= 3
			}
			multiply = {
				value = 0.0
				desc = "from_three_or_more_prior_members"
			}
		}
	}
}
