# Party for IG's that want free trade when it is not passed yet
# Should usually only appear when the Landowners want Free Trade

free_trade_party = {
	color = { 240 110 60 }
	name = {
		desc = party_free_trade
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/free_trade.dds"
	}
	unlocking_technologies = {
		stock_exchange
	}
	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			OR = {
				py:free_trade_party ?= {
					party_has_members = yes
				}
				any_interest_group = {
					ig_counts_as_marginal = no
					OR = {
						is_interest_group_type = ig_industrialists
						leader ?= {
							has_ideology = ideology:ideology_market_liberal
						}
					}
				}
			}
		}
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}
		if = { # # industrialists/market liberals split from liberals as the economy modernises
			limit = {
				OR = {
					leader ?= {
						has_ideology = ideology:ideology_market_liberal
					}
					has_ideology = ideology:ideology_plutocratic
				}
			}
			add = {
				desc = "from_liberal_economy_laws"
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_laissez_faire
								has_law_or_variant = law_type:law_interventionism
							}
						}
					}
					add = 2
				}
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_free_trade
								has_law_or_variant = law_type:law_protectionism
							}
						}
					}
					add = 1
				}
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_tenant_farmers
								has_law_or_variant = law_type:law_commercialized_agriculture
							}
						}
					}
					add = 1
				}
				if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_slavery_banned
						}
					}
					add = 1
				}
			}
		}
		# powerful capitalists likely to form their own party
		if = {
			limit = {
				is_powerful = yes
				OR = {
					leader ?= {
						has_ideology = ideology:ideology_market_liberal
					}
					has_ideology = ideology:ideology_plutocratic
				}
				NOT = {
					owner = {
						has_law_or_variant = law_type:law_universal_suffrage
					}
				}
			}
			add = {
				value = 10
				desc = "from_powerful_capitalists"
			}
		}
		# weaker market liberals fall behind powerful industrialists
		if = {
			limit = {
				is_powerful = no
				owner = {
					ig:ig_industrialists ?= {
						is_powerful = yes
					}
				}
				leader ?= {
					has_ideology = ideology:ideology_market_liberal
				}
			}
			add = {
				value = 5
				desc = "from_powerful_capitalists_other"
			}
		}
		if = { # landowners and industrialists hate labor rights
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_regulatory_bodies
						has_law_or_variant = law_type:law_worker_protections
					}
				}
				law_enactment_stance = {
					law = law_type:law_no_workers_rights
					value > neutral
				}
			}
			add = {
				value = 5
				desc = "from_labor_rights"
			}
		}
		if = { # conservatives hate welfare
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_wage_subsidies
						has_law_or_variant = law_type:law_old_age_pension
					}
				}
				law_enactment_stance = {
					law = law_type:law_poor_laws
					value > neutral
				}
			}
			add = {
				value = 5
				desc = "from_welfare"
			}
		}
		if = { # capitalists hate children
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_child_labor_allowed }
				}
				law_enactment_stance = {
					law = law_type:law_child_labor_allowed
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_hates_children"
			}
		}
		if = { # capitalists want free trade
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_free_trade }
				}
				law_enactment_stance = {
					law = law_type:law_free_trade
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_no_free_trade"
			}
		}
		if = { # capitalists want to have power
			limit = {
				owner = {
					NOR = {
						has_law_or_variant = law_type:law_oligarchy
						has_law_or_variant = law_type:law_wealth_voting
					}
				}
				has_ideology = ideology:ideology_plutocratic
			}
			add = {
				value = 3
				desc = "from_no_oligarchy_wealth_voting"
			}
		}
		if = { # liberal-capitalist anti-autocracy alliance
			limit = {
				owner = {
					has_law_or_variant = law_type:law_autocracy
				}
				law_enactment_stance = {
					law = law_type:law_wealth_voting
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_autocracy"
			}
		}
		if = { # capitalist-reactionary anti-democracy alliance
			limit = {
				owner = {
					has_law_or_variant = law_type:law_universal_suffrage
				}
				law_stance = {
					law = law_type:law_universal_suffrage
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_universal_suffrage"
			}
		}
		if = { # capitalist-meritocratic taxation alliance
			limit = {
				owner = {
					has_technology_researched = currency_standards
					NOT = { has_law_or_variant = law_type:law_per_capita_based_taxation }
				}
				law_enactment_stance = {
					law = law_type:law_per_capita_based_taxation
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_no_per_capita_tax"
			}
		}
		if = { # capitalist-conservative taxation alliance
			limit = {
				owner = {
					has_law_or_variant = law_type:law_graduated_taxation
				}
				law_stance = {
					law = law_type:law_graduated_taxation
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_graduated_taxation"
			}
		}
		if = { # opposition to command economy
			limit = {
				owner = {
					has_technology_researched = central_planning
					has_law_or_variant = law_type:law_command_economy
				}
				law_stance = {
					law = law_type:law_command_economy
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_command_economy"
			}
		}
		if = { # pro-private schools
			limit = {
				owner = {
					has_technology_researched = academia
					NOT = {
						has_law_or_variant = law_type:law_private_schools
					}
				}
				law_enactment_stance = {
					law = law_type:law_private_schools
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_lack_of_private_schools"
			}
		}
		if = { # protectionism bad
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_protectionist
				}
			}
			add = {
				value = -10
				desc = from_protectionist_leader
			}
		}
		if = { # universal suffrage Industrialists - prevents forming of impotent parties
			limit = {
				scope:number <= 1
				is_interest_group_type = ig_industrialists
				owner = {
					OR = {
						has_law_or_variant = law_type:law_universal_suffrage
						has_law_or_variant = law_type:law_census_voting
					}
				}
			}
			add = {
				desc = "from_late_industrialists"
				if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_universal_suffrage
						}
					}
					add = -10
				}
				if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_census_voting
						}
					}
					add = -5
				}
			}
		}
		if = {
			limit = {
				is_interest_group_type = ig_industrialists
				is_powerful = yes
			}
			multiply = {
				value = 2
				desc = "from_archetype_party"
			}
		}
		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				ig_counts_as_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = {
				scope:number = 2
			}
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number >= 3
			}
			multiply = {
				value = 0.0
				desc = "from_three_or_more_prior_members"
			}
		}
	}
}
