# Bastion of the Intelligenstia, also appeals to Industrialists
# Can sometimes sway PB and Rural Folk when they have common interests

liberal_party = {
	color = { 255 214 13 }
	name = {
		desc = party_liberal
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/liberal.dds"
	}
	unlocking_technologies = {
		empiricism
	}
	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			OR = {
				py:liberal_party ?= {
					party_has_members = yes
				}
				ig:ig_intelligentsia ?= {
					ig_counts_as_marginal = no
				}
				ig:ig_petty_bourgeoisie ?= {
					ig_counts_as_marginal = no
				}
				ig:ig_industrialists ?= {
					ig_counts_as_marginal = no
				}
			}
		}
		leader ?= { # far right and far left won't join liberal parties
			NOR = {
				has_ideology = ideology:ideology_fascist
				has_ideology = ideology:ideology_integralist
				has_ideology = ideology:ideology_ethno_nationalist
				has_ideology = ideology:ideology_communist_leader
				has_ideology = ideology:ideology_vanguardist
				has_ideology = ideology:ideology_anarchist
			}
		}
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {

		value = {
			value = -5
			desc = "base_value"
		}
		if = { # liberals love liberalism
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_reformer
						has_ideology = ideology:ideology_liberal_leader
						has_ideology = ideology:ideology_market_liberal
						has_ideology = ideology:ideology_republican_leader
						has_ideology = ideology:ideology_radical
					}
				}
			}
			add = {
				desc = "from_liberal_leader"
				if = {
					limit = {
						leader ?= {
							OR = {
								has_ideology = ideology:ideology_reformer
								has_ideology = ideology:ideology_liberal_leader
							}
						}
					}
					add = 10
				}
				if = {
					limit = {
						leader ?= {
							has_ideology = ideology:ideology_market_liberal
						}
					}
					add = 5
				}
				if = {
					limit = {
						leader ?= {
							OR = {
								has_ideology = ideology:ideology_republican_leader
								has_ideology = ideology:ideology_radical
							}
						}
					}
					add = 3
				}
			}
		}
		if = { # various progressives are inclined towards liberalism
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_feminist
						has_ideology = ideology:ideology_abolitionist
						has_ideology = ideology:ideology_pacifist
						has_ideology = ideology:ideology_humanitarian
						has_ideology = ideology:ideology_humanitarian_royalist
					}
				}
			}
			add = {
				value = 5
				desc = "from_progressivism"
			}
		}
		if = { # modernizer
			limit = {
				has_ideology = ideology:ideology_modernizer # The IG ideology
			}
			add = {
				value = 10
				desc = "from_modernizer"
			}
		}
		if = { # various conservatives dislike liberalism
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_traditionalist
						has_ideology = ideology:ideology_royalist
						has_ideology = ideology:ideology_theocrat
						has_ideology = ideology:ideology_slaver
						has_ideology = ideology:ideology_authoritarian
					}
				}
			}
			add = {
				value = -5
				desc = "from_conservatism"
			}
		}
		if = { # against hereditary bureacrats
			limit = {
				owner = {
					has_law_or_variant = law_type:law_hereditary_bureaucrats
				}
				law_stance = {
					law = law_type:law_hereditary_bureaucrats
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_hereditary_offices"
			}
		}
		if = { # for greater suffrage
			limit = {
				law_enactment_stance = {
					law = law_type:law_census_voting
					value > neutral
				}
			}
			if = {
				limit = {
					owner ?= {
						has_law = law_type:law_landed_voting
					}
				}
				add = {
					value = 5
					desc = "from_increased_suffrage"
				}
			}
			if = {
				limit = {
					owner ?= {
						has_law = law_type:law_wealth_voting
					}
				}
				add = {
					value = 3
					desc = "from_increased_suffrage"
				}
			}
			if = {
				limit = {
					owner ?= {
						has_law = law_type:law_census_voting
					}
				}
				add = {
					value = 1
					desc = "from_increased_suffrage"
				}
			}
		}
		if = { # armed forces will join if a more powerful member has jingoist leader
			limit = {
				is_interest_group_type = ig_armed_forces
				exists = owner.py:liberal_party
				owner = {
					any_interest_group = {
						is_member_of_party = owner.py:liberal_party
						is_powerful = yes
						leader ?= {
							has_ideology = ideology:ideology_jingoist_leader
						}
					}
				}
			}
			add = {
				value = 5
				desc = "from_jingoist_party_leader"
			}
		}
		# for as long as the Industrialists are not powerful, they'll be more liberal
		if = {
			limit = {
				is_powerful = no
				OR = {
					leader ?= {
						has_ideology = ideology:ideology_market_liberal
					}
					has_ideology = ideology:ideology_laissez_faire
					has_ideology = ideology:ideology_interventionist
				}
			}
			add = {
				value = 5
				desc = "from_capitalism"
			}
		}
		# If conservatives [landowners/church] are strong, Industrialists will stay liberal
		if = {
			limit = {
				NOR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_landowners
				}
				owner = {
					OR = {
						ig:ig_landowners ?= {
							is_powerful = yes
						}
						ig:ig_devout ?= {
							is_powerful = yes
						}
					}
				}
				OR = {
					leader ?= {
						has_ideology = ideology:ideology_market_liberal
					}
					has_ideology = ideology:ideology_laissez_faire
					has_ideology = ideology:ideology_interventionist
				}
			}
			add = {
				desc = "from_powerful_conservatives"
				if = {
					limit = {
						owner = {
							ig:ig_landowners ?= {
								is_powerful = yes
							}
						}
					}
					add = 2
				}
				if = {
					limit = {
						owner = {
							ig:ig_devout ?= {
								is_powerful = yes
							}
						}
					}
					add = 2
				}
			}
		}
		if = { # against illiteracy and religious education
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_religious_schools
						has_law_or_variant = law_type:law_no_schools
					}
				}
				law_enactment_stance = {
					law = law_type:law_private_schools
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_no_or_religious_schools"
			}
		}
		if = { # lack of women's rights
			limit = {
				owner = {
					has_law_or_variant = law_type:law_no_womens_rights
				}
				law_stance = {
					law = law_type:law_no_womens_rights
					value < neutral
				}
			}
			add = {
				value = 3
				desc = from_no_womens_rights
			}
		}
		if = { # anti-racism
			limit = {
				owner = {
					OR = {
						has_law_or_variant = law_type:law_ethnostate
						has_law_or_variant = law_type:law_national_supremacy
						has_law_or_variant = law_type:law_racial_segregation
					}
				}
				law_enactment_stance = {
					law = law_type:law_cultural_exclusion
					value > neutral
				}
			}
			add = {
				value = 3
				desc = from_racism
			}
		}
		if = { # anti-censorship
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_censorship
						has_law_or_variant = law_type:law_outlawed_dissent
					}
				}
				law_stance = {
					law = law_type:law_censorship
					value < neutral
				}
			}
			add = {
				value = 3
				desc = from_censorship
			}
		}
		if = { # against arbitrary or militarized policing
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_local_police
						has_law_or_variant = law_type:law_militarized_police
					}
				}
				law_stance = {
					law = law_type:law_militarized_police
					value < neutral
				}
				law_enactment_stance = {
					law = law_type:law_dedicated_police
					value > neutral
				}
			}
			add = {
				value = 3
				desc = from_bad_police
			}
		}
		if = { # anticlerical
			limit = {
				law_enactment_stance = {
					law = law_type:law_total_separation
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_anticlerical"
			}
		}
		if = { # republican
			limit = {
				owner ?= {
					OR = {
						country_has_monarchy_law = yes
						has_law_or_variant = law_type:law_theocracy
					}
				}
				OR = {
					has_ideology = ideology:ideology_republican
					has_ideology = ideology:ideology_populist
					leader ?= {
						OR = {
							has_ideology = ideology:ideology_nihilist
							has_ideology = ideology:ideology_republican_leader
							has_ideology = ideology:ideology_radical
						}
					}
				}
			}
			add = {
				value = 3
				desc = "from_republican"
			}
		}
		if = { # anti-serfdom
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_serfdom
				}
				law_stance = {
					law = law_type:law_serfdom
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_serfdom"
			}
		}
		if = { # general anti-slavery
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_legacy_slavery
						has_law_or_variant = law_type:law_slave_trade
						has_law_or_variant = law_type:law_colonial_slavery
					}
				}
				law_enactment_stance = {
					law = law_type:law_slavery_banned
					value > neutral
				}
			}
			add = {
				value = 5
				desc = "from_slavery"
			}
		}
		if = { # early Trade Unions
			limit = {
				is_interest_group_type = ig_trade_unions
				owner = {
					NOT = {
						has_technology_researched = socialism
					}
				}
			}
			add = {
				desc = "from_early_trade_unions"
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_landed_voting
								has_law_or_variant = law_type:law_wealth_voting
							}
						}
					}
					add = 10
				}
				else_if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_census_voting
						}
					}
					add = 5
				}
			}
		}
		if = {
			limit = {
				is_interest_group_type = ig_intelligentsia
				is_powerful = yes
			}
			multiply = {
				value = 2
				desc = "from_archetype_party"
			}
		}
		if = {
			limit = {
				scope:number <= 1
				is_powerful = no
				ig_counts_as_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = {
				scope:number = 2
			}
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number >= 3
			}
			multiply = {
				value = 0.0
				desc = "from_three_or_more_prior_members"
			}
		}
	}
}
