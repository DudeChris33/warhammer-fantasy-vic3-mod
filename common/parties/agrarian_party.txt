# Fallback party for Rural Folk
# Occasionally used for an anti-serfdom alliance

agrarian_party = {
	color = { 64 173 59 }
	name = {
		desc = party_agrarian
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/agrarians.dds"
	}

	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			OR = {
				py:agrarian_party ?= {
					party_has_members = yes
				}
				ig:ig_rural_folk ?= {
					ig_counts_as_marginal = no
				}
			}
		}
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	# scope:founding_party - yes if the root IG would be founding this party rather than joining an existing one
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}
		if = { # a party for powerful farmers
			limit = {
				has_ideology = ideology:ideology_agrarian
				is_powerful = yes
			}
			add = {
				value = 10
				desc = "from_peasants"
			}
		}
		if = { # weak anti-serfdom types might fall in line behind strong farmers
			limit = {
				is_powerful = no
				owner = {
					has_law_or_variant = law_type:law_serfdom
				}
				law_stance = {
					law = law_type:law_serfdom
					value < neutral
				}
			}
			add = {
				value = 10
				desc = "from_powerful_farmers"
			}
		}
		if = { # pacifist anti-colonial alliance
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_no_colonial_affairs }
				}
				law_enactment_stance = {
					law = law_type:law_no_colonial_affairs
					value > neutral
				}
			}
			add = {
				value = 6
				desc = "from_pacifism"
			}
		}
		if = { # anti-immigration alliance
			limit = {
				owner = {
					has_law_or_variant = law_type:law_no_migration_controls
				}
				law_stance = {
					law = law_type:law_no_migration_controls
					value < neutral
				}
			}
			add = {
				value = 6
				desc = "from_migration_law"
			}
		}
		if = { # limited suffrage Rural Folk - prevents forming of impotent parties
			limit = {
				is_interest_group_type = ig_rural_folk
				owner = {
					NOT = { has_law_or_variant = law_type:law_universal_suffrage }
				}
			}
			add = {
				desc = "from_early_rural_folk"
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_landed_voting
								has_law_or_variant = law_type:law_wealth_voting
							}
						}
					}
					add = -10
				}
				if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_census_voting
						}
					}
					add = -5
				}
			}
		}
		if = { # against arbitrary or militarized policing
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_local_police
						has_law_or_variant = law_type:law_militarized_police
					}
				}
				law_enactment_stance = {
					law = law_type:law_no_police
					value > neutral
				}
			}
			add = {
				value = 3
				desc = from_bad_police
			}
		}
		if = { # against land-based taxation
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_land_based_taxation
				}
				law_stance = {
					law = law_type:law_land_based_taxation
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_land_tax"
			}
		}
		if = { # against free trade - peasant/socialist alliance
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_free_trade
				}
				law_stance = {
					law = law_type:law_free_trade
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_anti_free_trade"
			}
		}
		if = { # against laissez-faire - peasant/socialist alliance
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_laissez_faire
				}
				law_stance = {
					law = law_type:law_laissez_faire
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_anti_laissez_faire"
			}
		}
		if = { # anti-peasant levies
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_peasant_levies
				}
				law_stance = {
					law = law_type:law_peasant_levies
					value < neutral
				}
			}
			add = {
				value = 3
				desc = "from_peasant_levies"
			}
		}
		if = { # anti-serfdom
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_serfdom
				}
				law_stance = {
					law = law_type:law_serfdom
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_serfdom"
			}
		}
		if = { # opposition to debt slavery
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_debt_slavery
				}
				law_stance = {
					law = law_type:law_debt_slavery
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_debt_slavery"
			}
		}
		if = { # nobody will join if the rural folk don't join
			limit = {
				scope:founding_party = no
				NOT = { is_interest_group_type = ig_rural_folk }
				OR = {
					NOT = { exists = owner.py:agrarian_party }
					AND = {
						exists = owner.py:agrarian_party
						owner = {
							ig:ig_rural_folk ?= {
								NOT = { is_member_of_party = owner.py:agrarian_party }
							}
						}
					}
				}
			}
			add = {
				value = -20
				desc = "from_no_rural_folk_in_party" # this should probably never actually display in game because the party won't exist
			}
		}
		if = {
			limit = {
				is_interest_group_type = ig_rural_folk
				is_powerful = yes
			}
			multiply = {
				value = 2
				desc = "from_archetype_party"
			}
		}
		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				ig_counts_as_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = {
				scope:number = 2
			}
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number >= 3
			}
			multiply = {
				value = 0.0
				desc = "from_three_or_more_prior_members"
			}
		}
	}
}
