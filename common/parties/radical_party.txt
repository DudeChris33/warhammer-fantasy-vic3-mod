# Radicals and Republicans extremely attracted to this ideology
# Attracts weak leftists and liberals

radical_party = {
	color = { 240 100 120 }
	name = {
		desc = party_radicals
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/radical.dds"
	}
	unlocking_technologies = {
		egalitarianism
	}
	# THIS IS RUN ONLY ONCE AT GAME START
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			OR = {
				py:radical_party ?= {
					party_has_members = yes
				}			
				any_interest_group = {
					ig_counts_as_marginal = no
					OR = {
						leader ?= {
							has_ideology = ideology:ideology_radical
						}
						is_interest_group_type = ig_trade_unions
					}
				}
			}
		}
		NOR = {
			is_interest_group_type = ig_landowners
			is_interest_group_type = ig_devout
		}
		leader ?= { # fascists and conservatives hate radicals
			NOR = {
				has_ideology = ideology:ideology_fascist
				has_ideology = ideology:ideology_integralist
				has_ideology = ideology:ideology_ethno_nationalist
				has_ideology = ideology:ideology_traditionalist
				has_ideology = ideology:ideology_royalist
				has_ideology = ideology:ideology_theocrat
				has_ideology = ideology:ideology_slaver
				always = no  # [#10327] was: character_has_carlist_ideology = yes
				always = no  # [#10327] was: character_has_miguelist_ideology = yes
			}
		}
	}

	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}
		if = { # radicals love radicalism
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_radical
				}
			}
			add = {
				value = 10
				desc = "from_radical"
			}
		}
		if = {
			limit = {
				is_powerful = yes
				leader ?= {
					has_ideology = ideology:ideology_radical
				}
			}
			add = {
				value = 15
				desc = "from_powerful_radical"
			}
		}
		if = { # radicals will fall in line behind stronger liberals
			limit = {
				OR = {
					has_ideology = ideology:ideology_egalitarian
					leader ?= {
						has_ideology = ideology:ideology_radical
					}
				}
				is_powerful = no
				owner = {
					any_interest_group = {
						is_powerful = yes
						OR = {
							has_ideology = ideology:ideology_liberal
							has_ideology = ideology:ideology_universalist
							has_ideology = ideology:ideology_liberal_modern
							has_ideology = ideology:ideology_universalist_modern
						}
					}
				}
			}
			add = {
				value = -5
				desc = "from_weak_radicals"
			}
		}
		if = { # weak liberals will fall in line behind strong radicals
			limit = {
				is_powerful = no
				OR = {
					has_ideology = ideology:ideology_liberal
					has_ideology = ideology:ideology_universalist
					has_ideology = ideology:ideology_liberal_modern
					has_ideology = ideology:ideology_universalist_modern
				}
				owner = {
					any_interest_group = {
						is_powerful = yes
						leader ?= {
							has_ideology = ideology:ideology_radical
						}
					}
				}
			}
			add = {
				value = 10
				desc = "from_weak_liberals_radical"
			}
		}
		if = { # weak leftists will fall in line behind radicals pre-revolution
			limit = {
				is_powerful = no
				leader ?= {
					has_socialist_ideology = yes
				}
				owner = {
					NOT = { has_law_or_variant = law_type:law_council_republic }
					any_interest_group = {
						is_powerful = yes
						leader ?= {
							has_ideology = ideology:ideology_radical
						}
					}
				}
			}
			add = {
				value = 10
				desc = "from_weak_leftists_radical"
			}
		}
		if = { # various progressives are inclined towards radicalism
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_feminist
						has_ideology = ideology:ideology_abolitionist
						has_ideology = ideology:ideology_state_socialist
						has_ideology = ideology:ideology_republican_leader
						has_ideology = ideology:ideology_nihilist
						has_ideology = ideology:ideology_humanitarian
					}
				}
			}
			add = {
				value = 10
				desc = "from_progressives_radical"
			}
		}
		if = { # early Trade Unions
			limit = {
				is_interest_group_type = ig_trade_unions
				owner = {
					NOT = {
						has_technology_researched = socialism
					}
				}
			}
			add = {
				desc = "from_early_trade_unions"
				if = {
					limit = {
						owner = {
							OR = {
								has_law_or_variant = law_type:law_landed_voting
								has_law_or_variant = law_type:law_wealth_voting
							}
						}
					}
					add = 15
				}
				else_if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_census_voting
						}
					}
					add = 10
				}
				else_if = {
					limit = {
						owner = {
							has_law_or_variant = law_type:law_universal_suffrage
						}
					}
					add = 5
				}
			}
		}
		if = { # republican
			limit = {
				owner ?= {
					OR = {
						country_has_monarchy_law = yes
						has_law_or_variant = law_type:law_theocracy
					}
				}
				OR = {
					has_ideology = ideology:ideology_republican
					has_ideology = ideology:ideology_populist
					leader ?= {
						OR = {
							has_ideology = ideology:ideology_nihilist
							has_ideology = ideology:ideology_republican_leader
							has_ideology = ideology:ideology_radical
						}
					}
				}
			}
			add = {
				value = 5
				desc = "from_republican"
			}
		}
		if = {
			limit = { # particularists will join in when their issues are relevant
				owner = {
					OR = {
						has_law_or_variant = law_type:law_serfdom
						has_law_or_variant = law_type:law_debt_slavery
					}
				}
				has_ideology = ideology:ideology_particularist
			}
			add = {
				value = 3
				desc = "from_particularism_cares_about_laws"
			}
		}
		if = { # moderates are scared of radical politics
			limit = {
				NOT = { is_interest_group_type = ig_trade_unions }
				leader ?= {
					has_ideology = ideology:ideology_moderate
				}
			}
			add = {
				value = -3
				desc = "from_moderate"
			}
		}
		if = { # from lack of democracy
			limit = {
				owner = {
					NOT = { has_law_or_variant = law_type:law_universal_suffrage }
				}
				law_enactment_stance = {
					law = law_type:law_universal_suffrage
					value > neutral
				}
			}
			add = {
				value = 5
				desc = "from_no_universal_suffrage"
			}
		}
		if = { # anti-censorship
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_censorship
						has_law_or_variant = law_type:law_outlawed_dissent
					}
				}
				law_stance = {
					law = law_type:law_censorship
					value < neutral
				}
			}
			add = {
				value = 3
				desc = from_censorship
			}
		}
		if = { # opposition to secret police
			limit = {
				owner = {
					has_law_or_variant = law_type:law_secret_police
				}
				law_stance = {
					law = law_type:law_secret_police
					value < neutral
				}
			}
			add = {
				value = 6
				desc = from_bad_home_affairs
			}
		}
		if = { # general anti-slavery
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_legacy_slavery
						has_law_or_variant = law_type:law_slave_trade
						has_law_or_variant = law_type:law_colonial_slavery
					}
				}
				law_enactment_stance = {
					law = law_type:law_slavery_banned
					value > neutral
				}
			}
			add = {
				value = 3
				desc = "from_slavery"
			}
		}
		if = { # anti-serfdom
			limit = {
				owner ?= {
					has_law_or_variant = law_type:law_serfdom
				}
				law_stance = {
					law = law_type:law_serfdom
					value < neutral
				}
			}
			add = {
				value = 5
				desc = "from_serfdom"
			}
		}
		if = {
			limit = {
				scope:number = 1
				is_powerful = no
				ig_counts_as_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = {
				scope:number = 2
			}
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number >= 3
			}
			multiply = {
				value = 0.0
				desc = "from_three_or_more_prior_members"
			}
		}
	}
}
