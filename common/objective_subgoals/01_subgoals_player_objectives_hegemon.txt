sg_colonization_laws = {
	category = sgcat_hegemon
	is_repeatable = no

	trigger = {
	}

	on_start = {
		# Deviation from vanilla: the unrecognized test runs FIRST. Vanilla checks
		# `country_rank <= rank_value:minor_power` before it checks unrecognized status,
		# so an unrecognized country below minor_power rank falls into the minor branch
		# and is handed sg_great_power, a subgoal it cannot reach. Testing country type
		# before rank sends every unrecognized country to sg_unrecognized_power instead.
		# The four variable names are vanilla's and must not change -- sg_greatest_power,
		# sg_greater_power, sg_great_power and sg_unrecognized_power gate on them.
		if = {
			limit = {
				ROOT = {
					is_country_type = unrecognized
				}
			}
			set_variable = objective_country_unrecognized
		}

		else_if = {
			limit = {
				ROOT = {
					country_rank = rank_value:great_power
				}
			}
			set_variable = objective_country_great
		}

		else_if = {
			limit = {
				ROOT = {
					country_rank = rank_value:major_power
				}
			}
			set_variable = objective_country_major
		}

		else_if = {
			limit = {
				ROOT = {
					country_rank <= rank_value:minor_power
				}
			}
			set_variable = objective_country_minor
		}

		add_journal_entry = {
			type = je_colonization_laws
			objective_subgoal = sg_colonization_laws
		}
	}
}
