# Country formation — Bretonnia. Board item #3417 (epic #3414).
# BRE (Bretonnia) is the region's only formable; every other 03_brettonia tag is a start tag.
#
# Deliberately minor/culture-based in shape: no unification_play, leadership_play or
# geographic_region key. common/diplomatic_plays and common/geographic_regions are blanked via
# .metadata/metadata.json's replace_paths and hold no entries, so referencing either would point at
# content that does not exist. Revisit when those databases are authored.
# Triggers used below are confirmed in docs/triggers.log (both country scope):
#   is_country_type = <country type>
#   country_has_primary_culture = <culture>

# bretonnian ONLY: the mousillon culture is deliberately excluded, so the cursed dukedom
# can never form the kingdom (parallel to EMP excluding sylvanian_vampire).

BRE = {
	use_culture_states = yes
	required_states_fraction = 0.75
	ai_will_do = { always = yes }

	possible = {
		is_country_type = recognized
		country_has_primary_culture = cu:bretonnian
	}
}

# ALN (Athel Loren): explicit state list, not use_culture_states, because wood_elf is also the
# homeland culture of Gryphon Wood, Laurelorn, Nuja and others far outside the forest. The thirteen
# Athel Loren states (the realms' states plus the glades), 0.75 of them (10) must be held.
# The wood elves are unrecognized, so the gate is unrecognized (BRE's recognized gate would lock them out).
ALN = {
	states = {
		STATE_ANMYR
		STATE_ARGWYLON
		STATE_GLADE_OF_ETERNAL_MIDNIGHT
		STATE_GLADE_OF_ETERNAL_MOONLIGHT
		STATE_HALLS_OF_EQUOS
		STATE_KING_S_GLADE
		STATE_TALSYN
		STATE_TAL_ESTH
		STATE_TORGOVANN
		STATE_TYR_VANNA
		STATE_WATERFALL_PALACE
		STATE_WYDRIOTH
		STATE_YI_EDRI_ETERNOS
	}
	required_states_fraction = 0.75
	ai_will_do = { always = yes }

	possible = {
		is_country_type = unrecognized
		country_has_primary_culture = cu:wood_elf
	}
}
