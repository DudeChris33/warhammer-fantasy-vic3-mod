# Country formation — the Empire. Board item #3416 (epic #3414).
# Two formables: EMP (the human Electorates unify) and VEM (the von Carsteins conquer the whole
# Empire instead). Every other 02_empire tag is a start tag.
#
# Deliberately minor/culture-based in shape: no unification_play, leadership_play or
# geographic_region key. common/diplomatic_plays and common/geographic_regions are blanked via
# .metadata/metadata.json's replace_paths and hold no entries, so referencing either would point at
# content that does not exist. Revisit when those databases are authored.
# Triggers used below are confirmed in docs/triggers.log (both country scope):
#   is_country_type = <country type>
#   country_has_primary_culture = <culture>

EMP = {
	use_culture_states = yes
	required_states_fraction = 0.75
	ai_will_do = { always = yes }

	possible = {
		is_country_type = recognized
		OR = {
			country_has_primary_culture = cu:reiklander
			country_has_primary_culture = cu:middenlander
			country_has_primary_culture = cu:nordlander
			country_has_primary_culture = cu:hochlander
			country_has_primary_culture = cu:ostlander
			country_has_primary_culture = cu:ostermarker
			country_has_primary_culture = cu:talabeclander
			country_has_primary_culture = cu:stirlander
			country_has_primary_culture = cu:averlander
			country_has_primary_culture = cu:wissenlander
			country_has_primary_culture = cu:sollander
			country_has_primary_culture = cu:sylvanian
		}
	}
}

VEM = {
	use_culture_states = yes
	required_states_fraction = 0.75
	ai_will_do = { always = yes }

	possible = {
		is_country_type = unrecognized
		country_has_primary_culture = cu:sylvanian_vampire
	}
}
