# PARTIAL PORT (#10567) of vanilla 1.15.1 common/character_interactions/02_favorite_interactions.txt.
# The ruler's Favorite (State and Revolution, still gated on has_dlc_feature = ip5_content).
# Dropped branches, each marked "# [#10567] removed:" at its site:
# - appoint_favorite.actor_capacity (possible before 1.15.2, #11808) and revoke_favorite.possible: the rasputin1_active_var trigger_if (Rasputin event lock)
#   and the favorite_event_active_var trigger_if. Both variables are set only by vanilla events the mod blanks
#   (events/ is in replace_paths), so the guards could never fire and would trip the "used but never set" lint.
#   Restore the second one if favorite events are authored.
# - appoint_favorite.effect: the je_tsarpoints_master event trigger (tsarpoints_events.6) and the Rasputin/Alexei
#   faith-healing branch (RUS_* templates, modifier_faith_healing, je_tp_fav_rasputin variable).
# - appoint_favorite.ai_chance: tsarpoints +10, Rasputin +60, Witte +80, je_tp_pool_decrease_landowner_influence -100,
#   je_tp_tsar_crown_of_perfection +50, c:RUS/tsarpoints x0.25; ai_strategy_great_reforms dropped from the
#   progressive OR (the strategy is not defined in this mod).
# - revoke_favorite.ai_chance: ai_strategy_great_reforms from the progressive OR, the two je_tp_* +100 blocks,
#   Rasputin x0.01.
# In play: no Russia-specific weighting or events; generic eligibility, controversy, cooldowns and AI weights as vanilla.
appoint_favorite = {
	icon = "gfx/interface/icons/character_role_icons/favorite.dds"
	clicksound = "event:/SFX/UI/Military/command_grant"

	potential = {
		has_dlc_feature = ip5_content
		owner ?= scope:actor
		is_ruler_of_own_country = no
		is_heir_of_own_country = no
		NOT = { has_role = character_role_favorite }
	}

	show_requirements = yes

	actor_capacity = {
		can_have_favorites = yes
		# [#10567] removed: trigger_if on has_variable = rasputin1_active_var (rasputin1_active_tt)
		# [#10567] removed: trigger_if on has_variable = favorite_event_active_var (favorite_event_active_tt)
	}

	effect = {
		owner ?= {
			random_scope_character = {
				limit = {
					has_role = character_role_favorite
				}
				revoke_favorite_effect = yes
			}
		}
		if = {
			limit = {
				owner ?= {
					any_interest_group = {
						favorite_appointment_ig_penalty >= 1
					}
				}
			}
			custom_tooltip = appoint_favorite_controversy_tt
			owner ?= {
				every_interest_group = {
					limit = {
						favorite_appointment_ig_penalty >= 1
					}
					add_modifier = {
						name = modifier_controversial_favorite
						days = normal_modifier_time
						is_decaying = yes
						multiplier = favorite_appointment_ig_penalty
					}
				}
				every_interest_group = {
					limit = {
						favorite_appointment_ig_penalty >= 1
						favorite_appointment_ig_radicals >= 0.01
					}
					save_temporary_scope_as = controversial_favorite_ig
					owner = {
						add_radicals = {
							value = scope:controversial_favorite_ig.favorite_appointment_ig_radicals
							interest_group = scope:controversial_favorite_ig
						}
					}
				}
			}
		}
		custom_tooltip = {
			text = appoint_favorite_effect_tt
			add_character_role = favorite
		}
		if = {
			limit = {
				has_modifier = revoked_favorite
			}
			remove_modifier = revoked_favorite
		}
		# [#10567] removed: owner has_journal_entry = je_tsarpoints_master -> trigger_event tsarpoints_events.6
		# [#10567] removed: Rasputin faith healing for Alexei (has_template = RUS_grigoriy_rasputin, modifier_faith_healing)
	}

	cooldown = { days = 180 }

	ai_chance = {
		value = 0
		if = {
			limit = {
				owner ?= {
					NOT = {
						any_scope_character = {
							has_role = character_role_favorite
						}
					}
				}
				interest_group ?= {
					is_insurrectionary = no
				}
			}
			add = 5
			# [#10567] removed: +10 if owner has_journal_entry = je_tsarpoints_master
			if = {
				limit = {
					interest_group ?= {
						OR = {
							is_powerful = yes
							is_in_government = yes
						}
					}
				}
				add = 25
			}
			if = {
				limit = {
					OR = {
						is_interest_group_leader = yes
						has_role = character_role_general
						has_role = character_role_admiral
					}
				}
				add = 10
			}
			# [#10567] removed: +60 if has_template = RUS_grigoriy_rasputin
			if = {
				limit = {
					owner ?= {
						OR = {
							has_strategy = ai_strategy_reactionary_agenda
							has_strategy = ai_strategy_conservative_agenda
						}
					}
					OR = {
						is_interest_group_type = ig_landowners
						is_interest_group_type = ig_devout
						is_interest_group_type = ig_armed_forces
						has_ideology = ideology:ideology_traditionalist
						has_ideology = ideology:ideology_royalist
					}
				}
				add = 20
			}
			if = {
				limit = {
					owner ?= {
						OR = {
							has_strategy = ai_strategy_progressive_agenda
							# [#10567] removed: has_strategy = ai_strategy_great_reforms
						}
					}
					OR = {
						is_interest_group_type = ig_intelligentsia
						is_interest_group_type = ig_industrialists
						is_interest_group_type = ig_petty_bourgeoisie
					}
				}
				add = 20
			}
			if = {
				limit = {
					has_ideology = ideology:ideology_modernizer_leader
				}
				add = 20
			}
			# [#10567] removed: +80 if has_template = rus_witte_character_template
			# [#10567] removed: -100 for ig_landowners if owner has_journal_entry = je_tp_pool_decrease_landowner_influence
			# [#10567] removed: +50 for ig_intelligentsia if owner has_journal_entry = je_tp_tsar_crown_of_perfection
			# [#10567] removed: x0.25 for armed forces/generals/jingoists if owner is c:RUS or has je_tsarpoints_master
			subtract = {
				value = favorite_appointment_controversy
				multiply = 5
			}
		}
	}
}

revoke_favorite = {
	icon = "gfx/interface/icons/character_role_icons/favorite.dds"
	clicksound = "event:/SFX/UI/Military/commander_retire"

	potential = {
		has_dlc_feature = ip5_content
		owner ?= scope:actor
		has_role = character_role_favorite
	}

	show_requirements = yes

	possible = {
		current_career_length = {
			role = character_role_favorite
			months = 12
		}
		# [#10567] removed: trigger_if on owner has_variable = rasputin1_active_var (rasputin1_active_tt)
		# [#10567] removed: trigger_if on owner has_variable = favorite_event_active_var (favorite_event_active_tt)
	}

	effect = {
		revoke_favorite_effect = yes
	}

	cooldown = { days = normal_modifier_time }

	ai_chance = {
		value = 0
		if = {
			limit = {
				interest_group ?= {
					is_insurrectionary = yes
				}
			}
			add = 80
		}
		if = {
			limit = {
				owner ?= {
					OR = {
						has_strategy = ai_strategy_reactionary_agenda
						has_strategy = ai_strategy_conservative_agenda
					}
				}
				OR = {
					has_socialist_ideology = yes
					has_progressive_ideology = yes
					has_liberal_ideology = yes
					is_interest_group_type = ig_trade_unions
				}
			}
			add = 15
		}
		if = {
			limit = {
				owner ?= {
					OR = {
						has_strategy = ai_strategy_progressive_agenda
						has_strategy = ai_strategy_egalitarian_agenda
						# [#10567] removed: has_strategy = ai_strategy_great_reforms
					}
				}
				OR = {
					has_reactionary_ideology = yes
					has_ideology = ideology:ideology_traditionalist
					has_ideology = ideology:ideology_royalist
				}
			}
			add = 15
		}
		if = {
			limit = {
				interest_group ?= {
					ig_approval <= -10
				}
			}
			add = 5
		}
		# [#10567] removed: +100 for ig_landowners if owner has_journal_entry = je_tp_pool_decrease_landowner_influence
		# [#10567] removed: +100 for non-intelligentsia if owner has_journal_entry = je_tp_tsar_crown_of_perfection
		# [#10567] removed: x0.01 if has_template = RUS_grigoriy_rasputin
	}
}
