# common/coat_of_arms/coat_of_arms is a replace_paths folder, so vanilla's copy of
# this file does NOT load. Bespoke content per board item #3630, epic #3621 - the 6
# Dark Lands country tags (common/country_definitions/10_dark_lands.txt, same order).
# Composed from vanilla gfx/coat_of_arms/ art (patterns + colored_emblems, neither
# blanked by replace_paths), two Realms of Exether colored_emblems (reuse grant
# confirmed 2026-09-07, see epic #3621's Art Palette section for provenance-comment
# convention), and this mod's named_colors palette (common/named_colors/00_coa_colors.txt
# + 01_wh_coa_colors.txt). Requires common/flag_definitions/10_dark_lands.txt (this
# mod's DEFAULT flag_definitions block is unconditional at priority = 0 and would
# otherwise mask every one of these - see that file's header) to actually render
# in-game.

# ZHA - Dawi Zharr Empire, the pan-Chaos-Dwarf unification formable. Canon device
# is the Bull of Hashut; reserved for THIS tag rather than MZN (the actual canon
# holy city of Zharr-Naggrund) as a deliberate mechanical choice to avoid charge
# duplication between the two formables/tags, not an oversight.
# emblem: Realms of Exether (gfx/coat_of_arms/colored_emblems/ce_bull.dds), reuse grant 2026-09-07
ZHA = {
	pattern = "pattern_solid.tga"
	color1 = "black"
	color2 = "black"

	colored_emblem = {
		texture = "ce_bull.dds"
		color1 = "orange"
		color2 = "black"
		instance = { position = { 0.5 0.5 } scale = { 1 1 } }
	}
}

# MZN - Mingol Zharr-Naggrund, the ziggurat-city itself and the seat of Hashut
# worship. No ziggurat/pyramid/tower-temple art exists in vanilla OR the Realms of
# Exether set (confirmed by a targeted sweep during scoping) - a stepped
# triple-border pattern plus a castle emblem stand in for the tiered ziggurat
# architecture. Deliberately NOT the bull, which is reserved for ZHA above.

	pattern = "pattern_border_of_3.dds"

# neither of which is a star.
