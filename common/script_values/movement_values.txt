movement_radicalism_lack_of_suffrage = {
    value = 0
    if = {
        limit = {
            has_global_variable = global_democracy_index
            owner ?= {
                has_technology_researched = egalitarianism
                country_has_voting_franchise = no
            }
        }
        add = {
            desc = "LACK_OF_SUFFRAGE"
            value = global_var:global_democracy_index
            multiply = {
                value = owner.modifier:state_expected_sol_from_literacy
                multiply = 0.10
                add = 0.5
            }
            if = {
                limit = {
                    owner ?= {
                        has_law_or_variant = law_type:law_elected_bureaucrats
                    }
                }
                divide = 20
            }
            else = {
                divide = 15
            }
            min = 0
            max = 0.4
        }
    }
}
movement_radicalism_suppressed_after_civil_war = {
    value = 0
    if = {
        limit = {
            this.days_since_movement_defeated <= very_long_modifier_time
        }
        add = { # (x-y)/y - ([1, 7300]-7300)/7300
            desc = "movement_suppressed_after_civil_war"
            value = this.days_since_movement_defeated
            subtract = very_long_modifier_time
            divide = {
                value = very_long_modifier_time
            }
        }
    }
}

movement_radicalism_simultaneous_insurrections = {
    value = 0
    if = {
        limit = {
            owner ?= {
                any_political_movement = {
                    NOT = { this = root } # Insurrectionary movements should not make themselves more insurrectionary
                    is_insurrectionary = yes
                }
            }
        }
        add = {
            desc = "SIMULTANEOUS_INSURRECTIONARY_MOVEMENTS"
            owner ?= {
                every_political_movement = {
                    limit = {
                        NOT = { this = root } # Insurrectionary movements should not make themselves more insurrectionary
                        is_insurrectionary = yes
                    }
                    if = {
                        limit = {
                            exists = this.political_movement_identity_support
                        }
                        add = { # Identity support can get a bit high
                            value = this.political_movement_identity_support
                            multiply = 0.5
                        }
                    }
                    else = {
                        add = this.political_movement_support
                    }
                }
            }
            min = 0.05
            max = 0.25
        }
    }
}
movement_radicalism_royalist_monarchy_abolished = {
    value = 0
    owner ?= {
        if = {
            limit = {
                has_law = law_type:law_monarchy
            }

            if = {
                limit = {
                    OR = {
                        is_enacting_law = law_type:law_presidential_republic
                        is_enacting_law = law_type:law_parliamentary_republic
                        is_enacting_law = law_type:law_corporate_state
                        is_enacting_law = law_type:law_theocracy
                        is_enacting_law = law_type:law_council_republic
                    }
                }

                add = {
                    value = 0.25
                    desc = "MONARCHY_BEING_ABOLISHED"
                }
            }
            else = {
                add = { # Only care about legitimacy in the context of a monarchy that isn't under direct threat
                    value = 50
                    subtract = legitimacy
                    multiply = 0.005
                    desc = "concept_legitimacy"
                }
            }
        }
    }
}
