# Religious-heritage traits (type = heritage). Each religion's `heritage =` points at one of these.
# Faiths of one stock share one trait: the Old World pantheon's cults coexist, so inter-cult friction
# comes through the religion axis, not a heritage split (decision #2 / review panel 2026-08-28). A group may hold
# several traits where faiths are kin but hostile: the elven gods hold heritage_elven_pantheon (the five
# elven cults) and heritage_cytharai (Druchii), board item #8476. Board item #3405 (epic #3404).

heritage_old_world_pantheon = { type = heritage  trait_group = heritage_group_old_world_pantheon }
heritage_ancestor_cult = { type = heritage  trait_group = heritage_group_ancestor_gods }
heritage_chaos_worship = { type = heritage  trait_group = heritage_group_chaos }
heritage_greenskin_faith = { type = heritage  trait_group = heritage_group_greenskin_gods }
heritage_mortuary_rites = { type = heritage  trait_group = heritage_group_mortuary_cult }
heritage_nagashite = { type = heritage  trait_group = heritage_group_undeath }

# Board items #6379, #8475. heritage_elven_pantheon: the five elven cults (Asuryan, Isha, Vaul, Hoeth, Kurnous).
heritage_elven_pantheon = { type = heritage  trait_group = heritage_group_elven_gods }
heritage_great_maw_faith = { type = heritage  trait_group = heritage_group_great_maw }

# Board item #8476. The Druchii's Cytharai: same elven gods, own trait, so the Druchii share the
# elven group (partial tolerance) without the Asrai/Asur's shared trait (full tolerance).
heritage_cytharai = { type = heritage  trait_group = heritage_group_elven_gods }
