# Every other trigger stays an inert `always = no` stub with no reference to the blanked real-world
# cultures, tags or heritages: pops fall through to default clothing.
# Never overwrite this file with vanilla's. If vanilla's trigger set changes, add or remove names here.

# [#9046] Partial port of vanilla 00_clothes_triggers.txt:1085, restored for Nipponese rulers. Dropped
# the law_bakufu branch (shogunate emperor via special_character_japanese_emperor) and the
# gov_domain_alliance/gov_domain_frontier exclusions (real-world Japanese law and government types);
# the language_japanese / c:JAP culture test becomes culture = cu:nipponese. What remains: a Nipponese
# ruler of a monarchy. Shogun and restoration triggers stay `no`.
japanese_imperial_clothes_trigger = {
	exists = scope:character
	scope:character = {
		exists = home_country
		is_ruler_of_own_country = yes
		home_country ?= {
			OR = {
				has_law = law_type:law_monarchy
				has_law = law_type:law_social_monarchy
			}
		}
		culture = cu:nipponese
	}
}

# [#6639] WH-native replacement (not a port) of vanilla 00_clothes_triggers.txt:1453. Vanilla ORs
# three real-world branches, all dangling here: cu:japanese modernizers, the AST_william_light
# template, and heritage_group_european minus cu:circassian and Ottoman/Egyptian-serving cu:albanian
# characters. This mod gates on the culture's art family instead: true for any character whose
# culture declares graphics = european (monstrous cultures included; their portraits are human
# placeholders). Other graphics families keep default clothing.
# [#9046] Also true, by culture id, for the non-European-graphics cultures that have no restored
# national dress of their own: khureshi, hobgoblin, cathayan_orc, kurgan (east_asian) and lizardmen
# (decentralised_americas). Native-American and american_clothes triggers stay `no`.
european_clothes_trigger = {
	exists = scope:character
	scope:character = {
		OR = {
			culture = { has_culture_graphics = european }
			culture = cu:khureshi
			culture = cu:hobgoblin
			culture = cu:cathayan_orc
			culture = cu:kurgan
			culture = cu:lizardmen
		}
	}
}

# [#6639] Partial port of vanilla 00_clothes_triggers.txt:1732. Dropped `NOT = { culture = cu:griqua }`
# (real-world culture, not defined in this mod; the NOT was always true here). Prunes nothing in play.
monarchy_clothes_trigger = {
	exists = scope:character
	scope:character = {
		exists = owner
		OR = {
			AND = {
				OR = {
					is_female = yes # Female Regents wear more stylish clothes than male ones, as female regents trend towards being royals themselves
					owner ?= { has_gov_regency = no }
				}
				is_ruler_of_own_country = yes
			}
			is_heir_of_own_country = yes
		}
		owner = {
			OR = {
				has_law = law_type:law_monarchy
				has_law = law_type:law_social_monarchy
			}
		}
	}
}
