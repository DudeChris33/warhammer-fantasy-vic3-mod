# NOTE (board #3540, epic #3292): this trigger and the two below it originally
# reasoned about je_government_petition / je_government_petition_negotiation_version(_b)
# / je_law_imposition - REAL vanilla mechanics (power-bloc law imposition, interest-group
# petition negotiation), not flavor content. Their journal entries were never authored
# here (common/journal_entries is blanked/deferred per epic #3292's wholesale deferral),
# so has_journal_entry on them is a permanent false and every branch/AND that depended on
# them has been dropped per the port-vs-stub decision rule in .claude/rules/scripting.md.
# This is a disclosed gap: it strips the AI's ability to reason about active law-imposition
# and petition-negotiation journal entries via these three triggers specifically.
has_enactment_je_or_law_commitment = {
	OR = {
		has_any_law_commitment = yes
		# [#10327] removed (Russian JE agendas not in this mod): je_tsarpoints_has_law_agenda = yes
		# [#10327] removed (Russian JE agendas not in this mod): congress_poland_ai_has_law_agenda = yes
	}
}

ai_has_reasons_to_not_oppose_law = {
	OR = {
		has_law_commitment = scope:law.type
		any_political_movement = {
			law_enactment_stance = {
				law = scope:law.type
				value > neutral
			}
			OR = {
				political_movement_radicalism >= 0.75
				AND = {
					political_movement_radicalism >= 0.25
					any_influenced_interest_group = {
						is_in_government = yes
					}
					relevant_law_is_critical_to_modernization = yes
				}
			}
		}
	}
}

ai_has_enact_weight_modifier_journal_entries = {
	OR = {
		has_law_commitment = scope:law.type
	}
}

has_powerful_protectors = {
	exists = $ACTOR_SCOPE$
	any_diplomatically_relevant_country = {
		NOT = { this = $ACTOR_SCOPE$ }
		NOT = { is_in_same_power_bloc = $ACTOR_SCOPE$ }
		NOT = { is_subject_of = $ACTOR_SCOPE$ }
		OR = {
			country_rank >= $ACTOR_SCOPE$.country_rank
			army_size >= $ACTOR_SCOPE$.army_size
		}
		any_scope_treaty = {
			binds = root
			any_scope_article = {
				OR = {
					has_type = alliance
					has_type = defensive_pact
					has_type = guarantee_independence
				}
			}
		}
		OR = {
			can_reach_target_country_to_protect = { TARGET_COUNTRY = root }
			can_reach_target_country_to_conquer = { TARGET_COUNTRY = $ACTOR_SCOPE$ }
		}
	}
}

target_is_our_or_fellow_subject = {
	OR = {
		is_subject_of = root
		AND = {

		}
	}

ai_can_incorporate_state = {
	OR = {
		is_homeland_of_country_cultures = root
		primary_cultures_percent_state = {
			target_country = root
			value >= 0.5
		}
		AND = {
			primary_cultures_percent_state = {
				target_country = root
				value >= 0.25
			}
			any_neighbouring_state = {
				owner = root
				is_incorporated = yes
			}
		}
		"years_to_incorporate(root)" <= define:NAI|INCORPORATE_STATE_MAX_YEARS
	}
}

ai_will_incorporate_state = {
	state_population >= define:NAI|INCORPORATE_STATE_MIN_POPULATION
	ai_can_incorporate_state = yes
}

	OR = {
		is_country_type = unrecognized
		is_country_type = decentralized # Could be useful for the rare cases this is used outside of AI strategies
	}
	OR = {
		is_subject = no

			is_country_type = unrecognized
		}
	}
}

# common/ai_strategies/00_default_strategy.txt (#3294). Vanilla's definitions
# (00_ai_triggers.txt) are entirely real-world: every branch of each OR checks
# a specific historical journal entry, tag, culture, or has_variable flag
# (je_risorgimento, je_oregon, je_danubian_federation, c:RSM, c:GBR, c:RUS,
# cu:korean, AFG_appealed_to_GBR, sided_with_korean_gov, etc.) - none of which

# per the same disclosed-stub convention as is_in_guerilla_war
# (01_legacy_compat_triggers.txt, audit #3447).
#
# $TARGET$ must actually appear in the body or the engine rejects the call
# site's TARGET = ... argument ("Compiling source ... failed for unknown
# arguments: TARGET" / "Scripted trigger should have no arguments" - caught
# by an in-game load after the first stub draft omitted it). $TARGET$ = {
# always = yes } is a harmless, always-true use of the parameter on whatever
# scope the caller passes, so the always = no still governs the result.

ai_has_systemic_conquest_reason = {
	OR = {
		$TARGET$ = { # They hold states we claim
			any_scope_state = { has_claim_by = root }
		}

		$TARGET$ = { # They hold our cultural homelands
			any_scope_state = { is_homeland_of_country_cultures = root }
		}

		has_diplomatic_pact = { # Our Rival
			who = $TARGET$
			type = rivalry
		}

		AND = { # Larger nations competing with eachother
			has_overlapping_interests = $TARGET$
			country_rank >= rank_value:major_power
			$TARGET$ = { country_rank >= rank_value:major_power }
		}

		AND = { # Our glorious right to self-determination..
			is_subject_of = $TARGET$
			liberty_desire >= 50
		}

		AND = { # ..and our ungrateful rebellious subjects!
			$TARGET$ = {
				is_subject_of = root
				liberty_desire >= 50
			}
		}

		AND = { # Global pariah
			country_rank >= rank_value:great_power
			$TARGET$ = { infamy >= infamy_threshold:pariah }
		}

		ai_is_regional_objective_local_country = {
			TYPE = aro:ai_regional_objective_acquire_states
			TARGET = $TARGET$
		}
	}

	can_reach_target_country_to_conquer = { TARGET_COUNTRY = $TARGET$ }
}
ai_has_systemic_subjugation_reason = {
	country_rank > $TARGET$.country_rank
	$TARGET$.country_rank <= rank_value:minor_power

	OR = {
		$TARGET$ = { ai_country_is_valid_colonization_target = yes }
		AND = {
			has_overlapping_interests = $TARGET$
			relations:root <= relations_threshold:poor
			NOT = { is_in_customs_union_with = $TARGET$ }
			$TARGET$ = {
				is_subject = no
				NOT = { is_subject_of = root }
			}
		}
		AND = { # Rebellious subject
			$TARGET$ = {
				is_subject_of = root
				liberty_desire >= 50
			}
		}

		ai_is_regional_objective_subjugate_target = { TARGET = $TARGET$ }
	}

	can_reach_target_country_to_subjugate = { TARGET_COUNTRY = $TARGET$ }
}
ai_has_systemic_befriend_reason = {
	country_rank <= $TARGET$.country_rank

	OR = {
		AND = { # A stronger country we already have good relations with
			has_overlapping_interests = $TARGET$
			$TARGET$ = { relations:root >= relations_threshold:cordial }
		}
		any_rival_country = { # Potential ally against rival
			has_overlapping_interests = $TARGET$
		}
	}

	can_reach_target_country_to_protect = { TARGET_COUNTRY = $TARGET$ }
}
ai_has_systemic_protect_reason = {
	country_rank > $TARGET$.country_rank

	OR = {
		AND = { # A weaker independent country
			has_overlapping_interests = $TARGET$
			$TARGET$ = { is_subject = no }
		}
		any_rival_country = { # Potential buffer state against rival
			has_strategic_adjacency = $TARGET$
		}

		ai_is_regional_objective_protect_target = { TARGET = $TARGET$ }
	}

	can_reach_target_country_to_protect = { TARGET_COUNTRY = $TARGET$ }
}
