		}
	}

#################################################################
#####				TRAIN CULTURE TRIGGERS					#####
#################################################################
#
# Vanilla's bodies for these are entirely cu:<real-world culture> lists
# (british/north_german/french/russian/yankee/...) that do not exist in this
# mod, so they are restored as stubs under their original names rather than
# ported. They only ever selected a regional *visual variant* of the train
# model, so a permanently-false result falls through to the default variant:
# a true no-op. gfx_has_no_culture_trains_trigger is the vanilla fallback
# (NOR over that same list), so its faithful stub is "yes".

gfx_has_british_trains_trigger = {
	always = no
}

gfx_has_german_trains_trigger = {
	always = no
}

gfx_has_french_trains_trigger = {
	always = no
}

gfx_has_russian_trains_trigger = {
	always = no
}

gfx_has_american_trains_trigger = {
	always = no
}

gfx_has_no_culture_trains_trigger = {
	always = yes
}
