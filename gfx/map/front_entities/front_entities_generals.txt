military_infantry_carlist_general_diorama_entity = {
	entity = "military_infantry_carlists_general_entity" # 

	group = "general"
	
	trigger = {
		scope:country = { always = no } # [#3410] was: scope:country = { c:SPC ?= this }
		has_dlc_feature = ip4_cosmetics
	}
}

#################################################
###       			European     			  ###
#################################################

military_infantry_european_general_diorama_entity = {
	entity = "military_infantry_european_general_entity"

	group = "general"
	
	trigger = {
		scope:culture ?= {
			has_culture_graphics = european
		}
		scope:country = { 
			OR = {
				NOT = { always = no } # [#3410] was: NOT = { c:SPC ?= this }
				AND = {
					always = no # [#3410] was: c:SPC ?= this
					NOT = { has_dlc_feature = ip4_cosmetics }
				}
			}
		}
	}
}
